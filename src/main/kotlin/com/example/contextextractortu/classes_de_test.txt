{

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.List;

/**
 * Classe principale à tester (SUT)
 * Hérite de OperationBase et utilise des services externes
 */
public class Calculatrice extends OperationBase {

    // Champs propres à Calculatrice
    private final Historique historique;
    private final ValidateurCalcul validateur;
    private BigDecimal tauxTva;

    // Constructeur avec @Autowired (simulé)
    public Calculatrice(Historique historique, ValidateurCalcul validateur) {
        super("Calculatrice");
        this.historique = historique;
        this.validateur = validateur;
        this.tauxTva = new BigDecimal("0.20");
    }

    // Setter pour tauxTva
    public void setTauxTva(BigDecimal tauxTva) {
        this.tauxTva = tauxTva;
    }

    /**
     * Calcul complexe utilisant héritage, services externes et méthodes internes
     */
    public CalculResultat calculerPanier(List<Item> items, boolean appliquerTva) {
        // Appel à méthode externe (ValidateurCalcul)
        if (!validateur.validerItems(items)) {
            throw new IllegalArgumentException("Items invalides");
        }

        // Appel à méthode interne (héritée de OperationBase)
        BigDecimal sousTotal = calculerSousTotal(items);

        // Appel à méthode interne privée
        BigDecimal montantTva = calculerTva(sousTotal, appliquerTva);

        // Appel à méthode externe
        BigDecimal totalFinal = historique.enregistrerCalcul(items, sousTotal, montantTva);

        // Utilisation de méthode interne héritée
        String nomCalcul = getNomOperation();

        return new CalculResultat(nomCalcul, sousTotal, montantTva, totalFinal);
    }

    /**
     * Méthode interne privée - doit être incluse dans le contexte
     */
    private BigDecimal calculerTva(BigDecimal base, boolean appliquer) {
        if (!appliquer) {
            return BigDecimal.ZERO;
        }
        return base.multiply(tauxTva).setScale(2, RoundingMode.HALF_UP);
    }

    /**
     * Calcul avec gestion d'erreur interne
     */
    public BigDecimal calculerAvecArrondi(List<BigDecimal> valeurs, int decimales) {
        // Appel à méthode interne héritée
        if (validerValeurs(valeurs)) {
            return arrondiResultat(valeurs, decimales);
        }
        return BigDecimal.ZERO;
    }

    /**
     * Méthode statique utilitaire interne
     */
    public static boolean estNombreValid(double nombre) {
        return !Double.isNaN(nombre) && !Double.isInfinite(nombre);
    }

    @Override
    public void initialiserContexte() {
        // Initialisation du contexte de calcul
        this.tauxTva = new BigDecimal("0.20");
    }
}
}

OperationBase.java :
{

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.List;

/**
 * Classe mere (Super-class) - fait partie des "Internes"
 * Contient des methodes qui seront appelees par Calculatrice
 */
public abstract class OperationBase {

    private String nomOperation;
    protected int compteurAppels;

    public OperationBase(String nomOperation) {
        this.nomOperation = nomOperation;
        this.compteurAppels = 0;
    }

    /**
     * Methode interne publique - doit etre recuperee dans le contexte
     */
    public BigDecimal calculerSousTotal(List<Item> items) {
        compteurAppels++;
        BigDecimal total = BigDecimal.ZERO;
        for (Item item : items) {
            total = total.add(item.getPrix().multiply(new BigDecimal(item.getQuantite())));
        }
        return total.setScale(2, RoundingMode.HALF_UP);
    }

    /**
     * Methode interne protegee - doit etre recuperee
     */
    protected boolean validerValeurs(List<BigDecimal> valeurs) {
        if (valeurs == null || valeurs.isEmpty()) {
            return false;
        }
        for (BigDecimal val : valeurs) {
            if (val == null || val.compareTo(BigDecimal.ZERO) < 0) {
                return false;
            }
        }
        return true;
    }

    /**
     * Methode interne de formattage
     */
    protected BigDecimal arrondiResultat(List<BigDecimal> valeurs, int decimales) {
        BigDecimal somme = BigDecimal.ZERO;
        for (BigDecimal val : valeurs) {
            somme = somme.add(val);
        }
        return somme.setScale(decimales, RoundingMode.HALF_UP);
    }

    /**
     * Getter pour nomOperation - methode interne simple
     */
    public String getNomOperation() {
        return nomOperation;
    }

    /**
     * Methode abstraite - les implementations externes ne sont pas connues
     */
    public abstract void initialiserContexte();
}
}
